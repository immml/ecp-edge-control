## 三、重装后恢复步骤（新系统装好后）

```bash
# 1) 基础依赖（Debian 12）
sudo apt update && sudo apt install -y curl unzip nmcli wireless-tools iw net-tools jq dnsutils bc

# 2) 恢复配置目录
sudo mkdir -p /opt/ecp-agent /etc/ecp-xray /home/orangepi/ecp-data /home/orangepi/ecp-config
# 把备份文件放回：
#   agent.yaml + ca.crt + client.crt + client.key + registration.key → /home/orangepi/ecp-config/
sudo chown -R orangepi:orangepi /home/orangepi/ecp-config /home/orangepi/ecp-data
sudo chmod 600 /home/orangepi/ecp-config/*.key

# 3) 安装 Agent（最新版，从云端 OTA 仓库拉）
sudo curl -fsSLk -o /opt/ecp-agent/ecp-agent \
  https://38.76.180.246:8443/api/v1/agent/binaries/ecp-agent-linux-arm64
sudo chmod +x /opt/ecp-agent/ecp-agent
# systemd 单元参考备份的 ecp-agent.service（ExecStart 改为 /opt/ecp-agent/ecp-agent）

# 4) 恢复 VPN 跳板（xray）
sudo curl -fsSL -o /opt/ecp-xray/xray \
  https://github.com/XTLS/Xray-core/releases/download/v26.3.27/Xray-linux-arm64-v8a.zip
# 或用 deploy/xray-vpn.sh --name orangepi3b 一键部署（会重新生成 UUID）
# xray-config.json 已备份，UUID=94d221ba-bbda-4036-9654-0c985af42e56

# 5) Tailscale
sudo tailscale up --hostname orangepi3b   # 重新登录 tailnet

# 6) 验证
systemctl status ecp-agent   # active
tailscale ip -4              # 100.108.234.5
# 控制面板应显示节点上线
```
